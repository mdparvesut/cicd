## 1. Secrets Detection in Pipeline

### 1.1 Why Secrets Detection?

**Common mistakes that leak secrets:**
```yaml
# ❌ BAD: Secret exposed in logs
script:
  - echo "Database password: ${DB_PASSWORD}"
  - echo "Connecting with password: mypassword123"
  
# ❌ BAD: Secret in Git history
git commit -m "Add config with password: admin123"

# ❌ BAD: Secret in code
const API_KEY = "sk-1234567890abcdef"
```

**Consequences:**
- 🔴 Attackers access your database
- 🔴 API keys stolen
- 🔴 Production compromised
- 🔴 Data breach

### 1.2 Install GitLab Secret Detection

GitLab has built-in secret detection. Enable it in your pipelines.

#### **Backend Pipeline - Add Secret Detection**

Edit `reddit-clone-backend/.gitlab-ci.yml` and add:

```yaml
# ========================================
# GitLab CI/CD Pipeline for Backend
# ========================================

stages:
  - security      # Add security stage FIRST
  - build
  - test
  - deploy

# ... (keep existing variables and cache)

# ========================================
# STAGE 0: SECURITY - Secret Detection
# Scan for exposed secrets in code
# ========================================
secret_detection:
  stage: security
  image: alpine:latest
  
  before_script:
    - apk add --no-cache git grep
  
  script:
    - echo "Scanning for exposed secrets..."
    
    # Check for common secret patterns
    - |
      echo "Checking for passwords in code..."
      if grep -r -i "password.*=.*['\"][^'\"]*['\"]" --include="*.go" --include="*.yaml" --include="*.yml" --exclude-dir=".git" . | grep -v "DB_PASSWORD\|JWT_SECRET\|placeholder\|example\|test"; then
        echo "❌ ERROR: Found hardcoded password in code!"
        exit 1
      fi
    
    - |
      echo "Checking for API keys..."
      if grep -r -E "(api[_-]?key|api[_-]?secret|access[_-]?token).*['\"][a-zA-Z0-9]{20,}['\"]" --include="*.go" --include="*.js" --exclude-dir=".git" .; then
        echo "❌ ERROR: Found potential API key in code!"
        exit 1
      fi
    
    - |
      echo "Checking for AWS keys..."
      if grep -r -E "AKIA[0-9A-Z]{16}" --include="*" --exclude-dir=".git" .; then
        echo "❌ ERROR: Found AWS access key!"
        exit 1
      fi
    
    - |
      echo "Checking for private keys..."
      if grep -r "BEGIN.*PRIVATE KEY" --include="*" --exclude-dir=".git" --exclude="*.md" .; then
        echo "❌ ERROR: Found private key in repository!"
        exit 1
      fi
    
    - |
      echo "Checking for database URLs with credentials..."
      if grep -r -E "(mysql|postgres|mongodb)://[^:]+:[^@]+@" --include="*.go" --include="*.yaml" --exclude-dir=".git" . | grep -v "example\|placeholder\|user:password"; then
        echo "❌ ERROR: Found database URL with credentials!"
        exit 1
      fi
    
    - echo "✅ No secrets detected in code"
  
  only:
    - branches
    - merge_requests

# ========================================
# STAGE 0.5: Check for secrets in logs
# ========================================
secrets_in_logs_check:
  stage: security
  image: alpine:latest
  
  variables:
    # Test variables that should NOT appear in logs
    TEST_SECRET: "this-should-be-masked"
  
  script:
    - echo "Testing secret masking..."
    
    # This should be masked by GitLab
    - echo "Masked variable test: ${TEST_SECRET}"
    
    # Verify masked variables don't leak
    - |
      if [ -n "${DB_PASSWORD}" ]; then
        echo "DB_PASSWORD is set (value is masked)"
      fi
    
    - |
      if [ -n "${JWT_SECRET}" ]; then
        echo "JWT_SECRET is set (value is masked)"
      fi
    
    # ❌ DON'T DO THIS IN REAL JOBS:
    # - echo "Password: ${DB_PASSWORD}"  # This would expose the secret!
    
    - echo "✅ Secret masking check completed"
  
  only:
    - branches
    - merge_requests

# ... (keep all existing stages: build, test, deploy)
```

// internal/utils/sanitize.go
package utils

import (
	"html"
	"strings"

	"github.com/microcosm-cc/bluemonday"
)

// SanitizeInput removes HTML and JavaScript from user input
func SanitizeInput(input string) string {
	// Trim whitespace
	input = strings.TrimSpace(input)

	// Escape HTML entities
	input = html.EscapeString(input)

	return input
}

// SanitizeHTML allows safe HTML but removes scripts
// Use for rich text content if needed
func SanitizeHTML(input string) string {
	// Create policy that allows safe HTML
	p := bluemonday.UGCPolicy()

	// Remove dangerous attributes
	p.AllowAttrs("class").Globally()

	// Sanitize
	return p.Sanitize(input)
}

// SanitizeStrictly removes ALL HTML
// Use for usernames, titles, etc.
func SanitizeStrictly(input string) string {
	p := bluemonday.StrictPolicy()
	return p.Sanitize(input)
}
```

func (h *PostHandler) CreatePost(c *gin.Context) {
	var req validation.CreatePostRequest

	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	// Validate input
	if err := req.Validate(); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})

	req.Content = utils.SanitizeHTML(req.Content)      // Allow safe HTML, remove scripts

	// Create post
	post, err := h.postService.CreatePost(req)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
		return
	}

	c.JSON(http.StatusCreated, post)
}

// internal/api/middleware/security.go
package middleware

import (
	"github.com/gin-gonic/gin"
)

// SecurityHeaders adds security headers to responses
func SecurityHeaders() gin.HandlerFunc {
	return func(c *gin.Context) {
		// Prevent clickjacking
		c.Header("X-Frame-Options", "DENY")

		// Prevent MIME type sniffing
		c.Header("X-Content-Type-Options", "nosniff")

		// Enable XSS protection in browser
		c.Header("X-XSS-Protection", "1; mode=block")

		// Content Security Policy - prevents inline scripts
		c.Header("Content-Security-Policy", "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data: https:; font-src 'self' data:;")

		// Strict Transport Security - force HTTPS
		c.Header("Strict-Transport-Security", "max-age=31536000; includeSubDomains")

		// Referrer policy
		c.Header("Referrer-Policy", "strict-origin-when-cross-origin")

		c.Next()
	}
}
```

      dangerouslySetInnerHTML={{ __html: post.content }} 
    />
    // This allows XSS if post.content contains <script>

// ✅ SAFE: Using DOMPurify
function SafeRichPost({ post }) {
  const sanitizeHTML = (html) => {
    return {
      __html: DOMPurify.sanitize(html, {
        ALLOWED_TAGS: ['b', 'i', 'em', 'strong', 'p', 'br'],
        ALLOWED_ATTR: []
      })
    };
  };

  return (
    <div dangerouslySetInnerHTML={sanitizeHTML(post.content)} />
  );
}

// src/utils/sanitize.js
import DOMPurify from 'dompurify';

/**
 * Sanitize HTML content to prevent XSS
 * @param {string} dirty - Potentially unsafe HTML
 * @returns {string} - Safe HTML
 */
export const sanitizeHTML = (dirty) => {
  return DOMPurify.sanitize(dirty, {
    ALLOWED_TAGS: ['b', 'i', 'em', 'strong', 'p', 'br', 'a', 'ul', 'ol', 'li'],
    ALLOWED_ATTR: ['href', 'target', 'rel'],
    ALLOW_DATA_ATTR: false,
  });
};

/**
 * Strip all HTML tags
 * @param {string} html - HTML content
 * @returns {string} - Plain text
 */
export const stripHTML = (html) => {
  return DOMPurify.sanitize(html, {
    ALLOWED_TAGS: [],
    ALLOWED_ATTR: [],
  });
};

/**
 * Sanitize user input (for forms)
 * @param {string} input - User input
 * @returns {string} - Sanitized input
 */
export const sanitizeInput = (input) => {
  // Trim whitespace
  let sanitized = input.trim();
  
  // Remove null bytes
  sanitized = sanitized.replace(/\0/g, '');
  
  // Escape HTML entities
  const div = document.createElement('div');
  div.textContent = sanitized;
  
  return div.innerHTML;
};
```

// internal/config/validate.go
package config

import (
	"errors"
	"fmt"
)

// Validate checks if configuration is valid
func (c *Config) Validate() error {
	// Check JWT secret
	if len(c.JWT.Secret) < 32 {
		return errors.New("JWT_SECRET must be at least 32 characters long")
	}

	if c.JWT.Secret == "CHANGE_ME" || c.JWT.Secret == "your-secret-here" {
		return errors.New("JWT_SECRET must be changed from default value")
	}

	// Check database password
	if c.Database.Password == "" {
		return errors.New("DB_PASSWORD cannot be empty")
	}

	if c.Database.Password == "CHANGE_ME" || c.Database.Password == "postgres" {
		return errors.New("DB_PASSWORD must be changed from default value")
	}

	// Check server mode
	if c.Server.Mode != "debug" && c.Server.Mode != "release" && c.Server.Mode != "test" {
		return fmt.Errorf("invalid GIN_MODE: %s", c.Server.Mode)
	}

	return nil
}

### 3.2 Secrets Rotation Strategy

**Document in README:**

```markdown
## Secrets Rotation

### When to Rotate Secrets

- ⚠️ **Immediately** if secret is exposed
- 🔄 **Quarterly** for JWT secrets
- 🔄 **Annually** for database passwords
- 🔄 **After** team member leaves

### How to Rotate JWT Secret

1. Generate new secret:
```bash
openssl rand -hex 32
```

2. Update in GitLab CI/CD Variables:
   - Settings → CI/CD → Variables
   - Edit `JWT_SECRET`
   - Save

3. Deploy will automatically use new secret

4. Note: All existing user sessions will be invalidated

### How to Rotate Database Password

1. Connect to database:
```bash

// internal/utils/logger.go
package utils

import (
	"regexp"
	"strings"
)

// SanitizeLog removes sensitive information from log messages
func SanitizeLog(message string) string {
	// Remove passwords
	re := regexp.MustCompile(`password[:\s]*[^\s]+`)
	message = re.ReplaceAllString(message, "password:***REDACTED***")

	// Remove JWT tokens
	re = regexp.MustCompile(`Bearer\s+[A-Za-z0-9-_=]+\.[A-Za-z0-9-_=]+\.?[A-Za-z0-9-_.+/=]*`)
	message = re.ReplaceAllString(message, "Bearer ***REDACTED***")

	// Remove API keys
	re = regexp.MustCompile(`(api[_-]?key|token)[:\s]*[A-Za-z0-9]{20,}`)
	message = re.ReplaceAllString(message, "$1:***REDACTED***")

	return message
}

// SafeLog logs a message after sanitizing sensitive data
func SafeLog(logger *log.Logger, level string, message string) {
	sanitized := SanitizeLog(message)
	logger.Printf("[%s] %s", level, sanitized)
}
