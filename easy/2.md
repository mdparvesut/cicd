## 💻 Part 4: Write Required Tests

### Test 1: Password Hashing (REQUIRED BY COURSE)

Create `backend/internal/auth/auth.go`:

```go
package auth

import (
    "golang.org/x/crypto/bcrypt"
)

// HashPassword hashes a plain text password
func HashPassword(password string) (string, error) {
    hashedBytes, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
    if err != nil {
        return "", err
    }
    return string(hashedBytes), nil
}

// VerifyPassword checks if the provided password matches the hash
func VerifyPassword(hashedPassword, password string) bool {
    err := bcrypt.CompareHashAndPassword([]byte(hashedPassword), []byte(password))
    return err == nil
}
```

Create `backend/internal/auth/auth_test.go`:

```go
package auth

import (
    "testing"
    "github.com/stretchr/testify/assert"
)

func TestHashPassword(t *testing.T) {
    password := "MySecurePassword123!"
    
    // Test that hashing works
    hash, err := HashPassword(password)
    
    assert.NoError(t, err, "Hashing should not return an error")
    assert.NotEmpty(t, hash, "Hash should not be empty")
    assert.NotEqual(t, password, hash, "Hash should be different from password")
}

func TestVerifyPassword_Success(t *testing.T) {
    password := "MySecurePassword123!"
    
    // Hash the password
    hash, err := HashPassword(password)
    assert.NoError(t, err)
    
    // Verify correct password
    isValid := VerifyPassword(hash, password)
    assert.True(t, isValid, "Correct password should verify successfully")
}

func TestVerifyPassword_Failure(t *testing.T) {
    password := "MySecurePassword123!"
    wrongPassword := "WrongPassword456!"
    
    // Hash the password
    hash, err := HashPassword(password)
    assert.NoError(t, err)
    
    // Verify wrong password
    isValid := VerifyPassword(hash, wrongPassword)
    assert.False(t, isValid, "Wrong password should not verify")
}

func TestHashPassword_DifferentHashesForSamePassword(t *testing.T) {
    password := "MySecurePassword123!"
    
    // Hash the same password twice
    hash1, err1 := HashPassword(password)
    hash2, err2 := HashPassword(password)
    
    assert.NoError(t, err1)
    assert.NoError(t, err2)
    
    // Hashes should be different (bcrypt uses random salt)
    assert.NotEqual(t, hash1, hash2, "Two hashes of same password should differ due to salt")
    
    // But both should verify correctly
    assert.True(t, VerifyPassword(hash1, password))
    assert.True(t, VerifyPassword(hash2, password))
}
```

### Test 2: Password Change Functionality (REQUIRED BY COURSE)

Create `backend/internal/models/user.go`:

```go
package models

import (
    "errors"
    "reddit-clone/internal/auth"
)

type User struct {
    ID       uint   `gorm:"primaryKey"`
    Username string `gorm:"unique;not null"`
    Password string `gorm:"not null"` // Hashed password
}

// ChangePassword changes the user's password
func (u *User) ChangePassword(currentPassword, newPassword string) error {
    // Verify current password
    if !auth.VerifyPassword(u.Password, currentPassword) {
        return errors.New("current password is incorrect")
    }
    
    // Validate new password (basic validation)
    if len(newPassword) < 8 {
        return errors.New("new password must be at least 8 characters")
    }
    
    // Hash new password
    hashedPassword, err := auth.HashPassword(newPassword)
    if err != nil {
        return err
    }
    
    // Update password
    u.Password = hashedPassword
    return nil
}
```

Create `backend/internal/models/user_test.go`:

```go
package models

import (
    "testing"
    "reddit-clone/internal/auth"
    "github.com/stretchr/testify/assert"
)

func TestChangePassword_Success(t *testing.T) {
    // Create a user with hashed password
    originalPassword := "OldPassword123!"
    hashedPassword, _ := auth.HashPassword(originalPassword)
    
    user := &User{
        ID:       1,
        Username: "testuser",
        Password: hashedPassword,
    }
    
    // Change password
    newPassword := "NewPassword456!"
    err := user.ChangePassword(originalPassword, newPassword)
    
    assert.NoError(t, err, "Password change should succeed")
    
    // Verify old password no longer works
    assert.False(t, auth.VerifyPassword(user.Password, originalPassword), 
        "Old password should not verify after change")
    
    // Verify new password works
    assert.True(t, auth.VerifyPassword(user.Password, newPassword), 
        "New password should verify after change")
}

func TestChangePassword_WrongCurrentPassword(t *testing.T) {
    originalPassword := "OldPassword123!"
    hashedPassword, _ := auth.HashPassword(originalPassword)
    
    user := &User{
        ID:       1,
        Username: "testuser",
        Password: hashedPassword,
    }
    
    // Try to change with wrong current password
    err := user.ChangePassword("WrongPassword!", "NewPassword456!")
    
    assert.Error(t, err, "Should return error for wrong current password")
    assert.Equal(t, "current password is incorrect", err.Error())
    
    // Password should remain unchanged
    assert.True(t, auth.VerifyPassword(user.Password, originalPassword))
}

func TestChangePassword_NewPasswordTooShort(t *testing.T) {
    originalPassword := "OldPassword123!"
    hashedPassword, _ := auth.HashPassword(originalPassword)
    
    user := &User{
        ID:       1,
        Username: "testuser",
        Password: hashedPassword,
    }
    
    // Try to change to a password that's too short
    err := user.ChangePassword(originalPassword, "short")
    
    assert.Error(t, err, "Should return error for short password")
    assert.Contains(t, err.Error(), "at least 8 characters")
}

### Test 3: Additional Business Logic Tests

Create `backend/internal/models/post_test.go`:

```go
package models

import (
    "testing"
    "github.com/stretchr/testify/assert"
)

type Post struct {
    ID        uint
    Title     string
    Content   string
    AuthorID  uint
    IsPrivate bool
    Upvotes   int
    Downvotes int
}

func (p *Post) GetScore() int {
    return p.Upvotes - p.Downvotes
}

func (p *Post) CanBeViewedBy(userID uint) bool {
    // Public posts can be viewed by anyone
    if !p.IsPrivate {
        return true
    }
    // Private posts can only be viewed by the author
    return p.AuthorID == userID
}

func TestGetScore(t *testing.T) {
    tests := []struct {
        name      string
        upvotes   int
        downvotes int
        expected  int
    }{
        {"positive score", 10, 3, 7},
        {"negative score", 3, 10, -7},
        {"zero score", 5, 5, 0},
        {"no votes", 0, 0, 0},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            post := &Post{
                Upvotes:   tt.upvotes,
                Downvotes: tt.downvotes,
            }
            
            score := post.GetScore()
            assert.Equal(t, tt.expected, score)
        })
    }
}

func TestCanBeViewedBy(t *testing.T) {
    authorID := uint(1)
    otherUserID := uint(2)
    
    t.Run("public post can be viewed by anyone", func(t *testing.T) {
        post := &Post{
            AuthorID:  authorID,
            IsPrivate: false,
        }
        
        assert.True(t, post.CanBeViewedBy(authorID), "Author should see public post")
        assert.True(t, post.CanBeViewedBy(otherUserID), "Other users should see public post")
    })
    
    t.Run("private post can only be viewed by author", func(t *testing.T) {
        post := &Post{
            AuthorID:  authorID,
            IsPrivate: true,
        }
        
        assert.True(t, post.CanBeViewedBy(authorID), "Author should see their private post")
        assert.False(t, post.CanBeViewedBy(otherUserID), "Other users should NOT see private post")
    })
}
```

###############################################

test-backend:
  stage: test
  image: golang:1.21-alpine
  
  before_script:
    # Install git (required for some go modules)
    - apk add --no-cache git
  
  script:
    - cd backend
    # Run tests with coverage
    - go test ./... -coverprofile=coverage.out -covermode=count
    
    # Calculate coverage percentage
    - |
      COVERAGE=$(go tool cover -func=coverage.out | grep total | awk '{print $3}' | sed 's/%//')
      echo "Test coverage: ${COVERAGE}%"
      
      # Check if coverage is at least 50%
      if (( $(echo "$COVERAGE < 50" | bc -l) )); then
        echo "ERROR: Coverage ${COVERAGE}% is below 50%"
        exit 1
      fi
  
  # Save coverage report as artifact
  artifacts:
    paths:
      - backend/coverage.out
    reports:
      coverage_report:
        coverage_format: cobertura
        path: backend/coverage.out
    expire_in: 1 week
  
  # Extract coverage for badge (this regex extracts the percentage)
  coverage: '/total:\s+\(statements\)\s+(\d+\.?\d*)%/'
  
  only:
    - main
    - merge_requests

## 🎨 Part 8: Customize Coverage Badge Colors

To change badge colors based on coverage thresholds, go to:

**GitLab** → **Settings** → **CI/CD** → **General pipelines** → **Test coverage parsing**

Add this regex:
```
total:\s+\(statements\)\s+(\d+\.?\d*)%
```

Then in **Badges** section:
- **Link**: `https://gitlab.com/%{project_path}/-/commits/%{default_branch}`
- **Badge image URL**: 
```
https://gitlab.com/%{project_path}/badges/%{default_branch}/coverage.svg?style=flat-square&min_good=50
```

This makes coverage:
- 🟢 **Green** if ≥50%
- 🟡 **Yellow** if 40-49%
- 🔴 **Red** if <40%

## 🎯 Understanding the Coverage Configuration

### The Magic Line:
```yaml
coverage: '/total:\s+\(statements\)\s+(\d+\.?\d*)%/'
```

This is a **regex pattern** that tells GitLab:
- Look for the line that says `total: (statements) XX.X%`
- Extract the number XX.X
- Display it as the coverage percentage

### Why We Check Coverage ≥50%:
```bash
if (( $(echo "$COVERAGE < 50" | bc -l) )); then
  echo "ERROR: Coverage ${COVERAGE}% is below 50%"
  exit 1
fi
```

This **fails the pipeline** if coverage drops below 50%, ensuring you maintain quality.

## 🐛 Troubleshooting

### Issue 1: "cannot find package golang.org/x/crypto/bcrypt"

**Solution:**
```bash
cd backend
go get golang.org/x/crypto/bcrypt
go mod tidy
git add go.mod go.sum
git commit -m "Add bcrypt dependency"
git push
```

### Issue 2: Tests pass locally but fail in pipeline

**Solution:** Make sure all dependencies are in `go.mod`:
```bash
cd backend
go mod tidy
```

### Issue 3: Coverage not showing in GitLab

**Solution:** 
1. Check the regex in `.gitlab-ci.yml` matches your output
2. Go to **Settings** → **CI/CD** → **General pipelines** → **Test coverage parsing**
3. Add the regex there too: `total:\s+\(statements\)\s+(\d+\.?\d*)%`

### Issue 4: "bc: command not found"

**Solution:** Update the alpine image:
```yaml
test-backend:
  stage: test
  image: golang:1.21-alpine
  
  before_script:
    - apk add --no-cache git bc
```
